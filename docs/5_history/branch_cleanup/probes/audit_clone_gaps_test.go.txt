package tests

// Audit-only verification tests for engine state-machine lifecycle review
// (2026-06-12). NOT intended for merge as-is — these tests document
// suspected coverage gaps in DeepCopy / RestoreFrom / ResetDynamicState.
// A test FAILURE here confirms the corresponding bug exists.

import (
	"testing"

	engine "gicg_mono/gicg_engine"
)

// Gap 1: DeepCopy does not carry Game.Preparing.
func TestAudit_DeepCopy_Preparing(t *testing.T) {
	env := NewGameWithDeck(t, []string{"赤蝶"}, []string{"刻师傅"})
	env.G.Preparing[0] = 12345

	clone := env.RT.Clone().Game
	if clone.Preparing[0] != 12345 {
		t.Errorf("DeepCopy dropped Preparing: got %d, want 12345", clone.Preparing[0])
	}
}

// Gap 2: RestoreFrom does not restore Game.Preparing.
func TestAudit_RestoreFrom_Preparing(t *testing.T) {
	env := NewGameWithDeck(t, []string{"赤蝶"}, []string{"刻师傅"})
	env.G.Preparing[1] = 777
	snap := env.G.DeepCopy()
	snap.Preparing[1] = 777 // simulate a snap that DOES carry it

	env.G.Preparing[1] = 0 // mutate live state
	env.G.RestoreFrom(snap)
	if env.G.Preparing[1] != 777 {
		t.Errorf("RestoreFrom dropped Preparing: got %d, want 777", env.G.Preparing[1])
	}
}

// Gap 3: RestoreFrom does not restore PlayerState.Supports.
func TestAudit_RestoreFrom_Supports(t *testing.T) {
	env := NewGameWithDeck(t, []string{"赤蝶"}, []string{"刻师傅"})
	snap := env.G.DeepCopy() // snapshot with 0 supports

	// Live game gains a support after the snapshot (speculative line).
	env.G.Players[0].Supports = append(env.G.Players[0].Supports,
		engine.SupportInst{Ref: 42, ActivatedAt: 1})

	env.G.RestoreFrom(snap)
	if n := len(env.G.Players[0].Supports); n != 0 {
		t.Errorf("RestoreFrom left %d phantom support(s) in zone, want 0", n)
	}
}

// Gap 4: ResetDynamicState does not clear Preparing nor Supports
// (cross-episode leak under engine reuse via capi GameReset).
func TestAudit_ResetDynamicState_PreparingSupports(t *testing.T) {
	env := NewGameWithDeck(t, []string{"赤蝶"}, []string{"刻师傅"})
	env.G.Preparing[0] = 999
	env.G.Players[1].Supports = append(env.G.Players[1].Supports,
		engine.SupportInst{Ref: 7, ActivatedAt: 2})

	env.RT.ResetDynamic(424242)

	if env.G.Preparing[0] != 0 {
		t.Errorf("ResetDynamicState leaked Preparing across episodes: got %d, want 0", env.G.Preparing[0])
	}
	if n := len(env.G.Players[1].Supports); n != 0 {
		t.Errorf("ResetDynamicState leaked %d support(s) across episodes, want 0", n)
	}
}

// Gap 5: CharEntry.SpecialtyCardRef is per-game dynamic state stored on
// the SHARED Ruleset CharRegistry — clones alias the same CharEntry, and
// ResetDynamic never resets it.
func TestAudit_SpecialtyCardRef_SharedAcrossClones(t *testing.T) {
	env := NewGameWithDeck(t, []string{"赤蝶"}, []string{"刻师傅"})

	cloneRT := env.RT.Clone()
	// Simulate the clone playing a Slot.Specialty card (builtins_card.go:206).
	cloneRT.Chars.BySlot[0][0].SpecialtyCardRef = 99

	if got := env.RT.Chars.BySlot[0][0].SpecialtyCardRef; got == 99 {
		t.Errorf("clone's specialty equip leaked into original runtime (shared CharEntry): got %d", got)
	}

	// And ResetDynamic does not clear it either.
	env.RT.Chars.BySlot[0][0].SpecialtyCardRef = 55
	env.RT.ResetDynamic(11)
	if got := env.RT.Chars.BySlot[0][0].SpecialtyCardRef; got != -1 {
		t.Errorf("ResetDynamic leaked SpecialtyCardRef across episodes: got %d, want -1", got)
	}
}

// Gap 6: deferred entries queued by on_after_damage hooks are dropped —
// the only DrainDeferred calls in DealDamage run BEFORE fireAfterDamage.
func TestAudit_DeferInAfterDamage_Dropped(t *testing.T) {
	env := NewGameWithDeck(t, []string{"赤蝶"}, []string{"刻师傅"})

	fired := false
	env.RT.Game.Hooks.Register(engine.Hook{
		Type: engine.HookAfterDamage,
		Fn: func(g *engine.Game, ctx *engine.EventContext) {
			g.Defer(func(g2 *engine.Game) { fired = true })
		},
	})

	hpID := env.RT.Chars.BySlot[1][0].HPCounterID
	env.G.DealDamage(hpID, engine.ElemPhysical, 1, engine.DamageOpts{ActorPlayer: 0, ActorChar: 0})

	if !fired {
		t.Errorf("deferred fn queued in on_after_damage was silently dropped")
	}
}
